function initBotEngine() { let pk = document.getElementById("bot-private-key").value.trim(); if (!pk) { alert("الرجاء إدخال المفتاح الخاص (Private Key) للمحفظة."); return; } if (!pk.startsWith("0x")) { pk = "0x" + pk; } try { provider = new ethers.providers.JsonRpcProvider(BSC_RPC); botWallet = new ethers.Wallet(pk, provider); // معالجة العنوان برمجياً ليكون مطابقاً لمعايير الـ Checksum النظامية safeAddress = ethers.utils.getAddress(botWallet.address); contract = new ethers.Contract(CONTRACT_ADDRESS, ERC20_ABI, botWallet); routerContract = new ethers.Contract(ROUTER_ADDRESS, ROUTER_ABI, botWallet); document.getElementById("engine-status").innerText = `● متصل: ${safeAddress.slice(0, 6)}...`; document.getElementById("engine-status").className = "text-xs font-mono bg-slate-950 px-4 py-2 rounded-2xl border border-slate-800 text-emerald-400"; logMessage("تم تشغيل محرك الأتمتة بنجاح للمحفظة: " + safeAddress, "success"); alert("تم ربط المحفظة بنجاح!"); } catch (err) { logMessage("خطأ في المفتاح الخاص: " + err.message, "error"); alert("خطأ في المفتاح الخاص: " + err.message); } }
