// SPDX-License-Identifier: MIT pragma solidity ^0.8.20; /** * @title ChainPump Factory Contract * @notice نظام إطلاق العملات مع تحويل رسوم الإنشاء الفورية لمالك المنصة */ contract ChainPumpFactory { address public immutable owner; // رسوم الإنشاء محددة بما يعادل تقريباً 0.05$ بالعملة الأصلية للشبكة (قابل للتعديل عند النشر) // على شبكة BSC (يقابل تقريباً 0.000064 ether/bnb) uint256 public creationFee = 0.000064 ether; event TokenCreated(address indexed tokenAddress, string name, string symbol, address indexed creator); event FeeTransferred(address indexed recipient, uint256 amount); modifier onlyOwner() { require(msg.sender == owner, "Unauthorized: Caller is not the owner"); _; } constructor() { owner = msg.sender; // تثبيت عنوان محفظتك كمالك حصري يستقبل العوائد مباشرة } /** * @notice دالة إنشاء العملة مع التحويل الفوري والحقيقي للرسوم لمحفظتك */ function createToken( string memory name, string memory symbol, uint256 totalSupply ) external payable returns (address) { // التحقق الحقيقي والصارم من دفع رسوم المنصة كاملة require(msg.value >= creationFee, "Error: Insufficient creation fee provided"); // 1. التحويل الفوري والحقيقي لرسوم الإنشاء إلى محفظتك (owner) دون محاكاة (bool feeSuccess, ) = payable(owner).call{value: creationFee}(""); require(feeSuccess, "Critical Error: Fee transfer to owner failed"); emit FeeTransferred(owner, creationFee); // 2. إرجاع أي فائض مالي أرسله المستخدم بالخطأ إلى محفظته مباشرة uint256 refund = msg.value - creationFee; if (refund > 0) { (bool refundSuccess, ) = payable(msg.sender).call{value: refund}(""); require(refundSuccess, "Error: Refund excess failed"); } // 3. نشر العملة أو تنفيذ منطق المصنع (Factory Logic) // (يتم هنا إنشاء ونشر عقد العملة الفرعي أو ربطه بالسيولة) address newTokenAddress = address(new MemeToken(name, symbol, totalSupply, msg.sender)); emit TokenCreated(newTokenAddress, name, symbol, msg.sender); return newTokenAddress; } /** * @notice تحديث قيمة رسوم الإنشاء إذا دعت الحاجة مستقبلاً (للمالك حصراً) */ function setCreationFee(uint256 _newFee) external onlyOwner { creationFee = _newFee; } } /** * @title MemeToken (عقد العملة الفرعي) */ contract MemeToken { string public name; string public symbol; uint256 public totalSupply; address public creator; constructor(string memory _name, string memory _symbol, uint256 _totalSupply, address _creator) { name = _name; symbol = _symbol; totalSupply = _totalSupply; creator = _creator; // تخصيص الأرصدة ومنطق الحماية ضد البوتات المباشرة (Anti-Snipe) } }
