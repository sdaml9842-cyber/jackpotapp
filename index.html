اضغط هنا للفتح المباشر في MetaMask 🚀` : `🦊 Open MetaMask App:

Click to open in MetaMask 🚀`, "error" ); return; } try { const accounts = await ethProvider.request({ method: "eth_requestAccounts" }); userAddress = accounts[0]; try { await ethProvider.request({ method: 'wallet_switchEthereumChain', params: [{ chainId: '0x2105' }], }); } catch (switchError) { if (switchError.code === 4902 || switchError.code === -32603) { await ethProvider.request({ method: 'wallet_addEthereumChain', params: [{ chainId: '0x2105', chainName: 'Base Mainnet', nativeCurrency: { name: 'ETH', symbol: 'ETH', decimals: 18 }, rpcUrls: ['https://mainnet.base.org'], blockExplorerUrls: ['https://basescan.org'] }] }); } } } catch (err) { console.error(err); showStatus(currentLang === 'ar' ?
