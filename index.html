متصفحك الحالي لا يدعم الإضافات. انقر لفتح المنصة مباشرة داخل تطبيق MetaMask:

فتح في تطبيق MetaMask 🚀` : `🦊 Open in MetaMask:
Click to open directly inside MetaMask app:

Open in MetaMask App 🚀`, "error" ); setTimeout(() => { window.location.href = mmDeepLink; }, 500); return; } try { const accounts = await ethProvider.request({ method: "eth_requestAccounts" }); userAddress = accounts[0]; try { await ethProvider.request({ method: 'wallet_switchEthereumChain', params: [{ chainId: '0x2105' }], }); } catch (switchError) { if (switchError.code === 4902 || switchError.code === -32603) { await ethProvider.request({ method: 'wallet_addEthereumChain', params: [{ chainId: '0x2105', chainName: 'Base Mainnet', nativeCurrency: { name: 'ETH', symbol: 'ETH', decimals: 18 }, rpcUrls: ['https://mainnet.base.org'], blockExplorerUrls: ['https://basescan.org'] }] }); } } } catch (err) { console.error(err); showStatus(currentLang === 'ar' ? "فشل الاتصال بمحفظة Base." : "Base connection failed.", "error"); return; } } if(userAddress) { const shortAddr = userAddress.substring(0, 6) + "..." + userAddress.substring(userAddress.length - 4); document.getElementById("connectWalletBtn").innerHTML = `${shortAddr}`; document.getElementById('referralLinkInput').value = `${window.location.origin}${window.location.pathname}?ref=${userAddress}`; showStatus(t.walletConnectedMsg, "success"); } }
